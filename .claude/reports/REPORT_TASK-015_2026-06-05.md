## Проблемы / открытые вопросы

- Vercel при Host=gpeters.ru отдаёт 404 (домен не привязан к проекту) — поэтому в proxy
  Host подменяется на `gala-antiage.vercel.app` (тогда 200). Зацикливания нет; добавлен
  `proxy_redirect`, чтобы vercel-домен не утекал в Location.
- `certbot renew --dry-run` в non-interactive режиме делает «random delay» (~6.5 мин) перед
  симуляцией — для быстрой проверки запускал с `--no-random-sleep-on-renew` (реальное
  авто-продление через systemd-таймер от этого не страдает).
- Production-домен Vercel определён и проверен curl'ом до прописывания: `gala-antiage.vercel.app`.

## Полный конфиг Nginx (`/etc/nginx/sites-available/gpeters.ru`, после certbot)

```nginx
server {
    server_name gpeters.ru www.gpeters.ru;

    # Приём форм: проксируем на локальный API, отдаём реальный IP посетителя.
    location /api/ {
        proxy_pass http://127.0.0.1:3001;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Витрина: всё остальное — на Vercel, Host подменяем на vercel-домен проекта.
    location / {
        proxy_pass https://gala-antiage.vercel.app;
        proxy_http_version 1.1;
        proxy_set_header Host gala-antiage.vercel.app;
        proxy_ssl_server_name on;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_redirect https://gala-antiage.vercel.app/ /;
    }

    listen [::]:443 ssl ipv6only=on; # managed by Certbot
    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/gpeters.ru/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/gpeters.ru/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot
}

server {
    if ($host = www.gpeters.ru) { return 301 https://$host$request_uri; } # managed by Certbot
    if ($host = gpeters.ru) { return 301 https://$host$request_uri; } # managed by Certbot
    listen 80;
    listen [::]:80;
    server_name gpeters.ru www.gpeters.ru;
    return 404; # managed by Certbot
}
```
